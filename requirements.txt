# Runtime dependencies for Meno.
# Install with:  pip install -r requirements.txt

# The headless wheel is required: opencv-python bundles its own Qt plugins,
# which conflict with PySide6's in the same process. The GUI is Qt's, so
# OpenCV's own windows are not used.
opencv-python-headless>=4.5   # capture, HSV, contours
numpy>=1.21                   # frame buffers and array math
PySide6-Essentials>=6.7       # desktop front end
